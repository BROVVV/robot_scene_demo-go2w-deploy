cmake_minimum_required(VERSION 3.8)
project(go2w_state_transport)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(unitree_go REQUIRED)
# Reuse the already bundled Unitree ROS header, also used by motion_control.
set(JSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../external/unitree_ros2/example/src/include")
if(NOT EXISTS "${JSON_INCLUDE}/nlohmann/json.hpp")
  message(FATAL_ERROR "The complete RobotScene source with bundled Unitree JSON header is required")
endif()
add_library(go2w_state_codec STATIC src/state_codec.cpp)
target_compile_features(go2w_state_codec PUBLIC cxx_std_17)
target_include_directories(go2w_state_codec PUBLIC include PRIVATE "${JSON_INCLUDE}")
foreach(node IN ITEMS network_low_state_capture local_low_state_relay)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PRIVATE cxx_std_17)
  target_link_libraries(${node} go2w_state_codec)
  ament_target_dependencies(${node} rclcpp unitree_go)
endforeach()
install(TARGETS go2w_state_codec network_low_state_capture local_low_state_relay
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_state_codec test/test_state_codec.cpp)
  target_compile_features(test_state_codec PRIVATE cxx_std_17)
  target_link_libraries(test_state_codec go2w_state_codec)
endif()
ament_package()
