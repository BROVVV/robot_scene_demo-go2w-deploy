cmake_minimum_required(VERSION 3.5)
project(go2w_nav2_compat)
find_package(ament_cmake REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
if(NOT "$ENV{ROS_DISTRO}" STREQUAL "foxy")
  message(FATAL_ERROR "This compatibility overlay is for Foxy only.")
endif()
add_library(go2w_footprint_compat SHARED src/footprint_subscriber.cpp src/odometry_utils.cpp)
target_include_directories(go2w_footprint_compat PUBLIC include)
ament_target_dependencies(go2w_footprint_compat nav2_util nav_msgs nav2_costmap_2d rclcpp tf2_ros tf2_geometry_msgs geometry_msgs)
install(TARGETS go2w_footprint_compat LIBRARY DESTINATION lib)
ament_export_libraries(go2w_footprint_compat)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(nav2_util nav_msgs nav2_costmap_2d rclcpp tf2_ros tf2_geometry_msgs geometry_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_footprint_timestamp test/test_footprint_timestamp.cpp)
  target_link_libraries(test_footprint_timestamp go2w_footprint_compat)
  ament_target_dependencies(test_footprint_timestamp nav2_util nav_msgs nav2_costmap_2d rclcpp tf2_ros tf2_geometry_msgs geometry_msgs)
  ament_add_gtest(test_required_parameters test/test_required_parameters.cpp)
  target_include_directories(test_required_parameters PRIVATE include)
  ament_target_dependencies(test_required_parameters nav2_util rclcpp tf2_ros tf2_geometry_msgs geometry_msgs)
endif()
ament_package()
