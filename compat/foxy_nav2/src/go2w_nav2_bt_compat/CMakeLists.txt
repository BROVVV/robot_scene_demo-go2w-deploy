cmake_minimum_required(VERSION 3.16)
project(go2w_nav2_bt_compat)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(go2w_foxy_bt_actions SHARED plugins/action/compute_path_to_pose_action.cpp
  plugins/action/follow_path_action.cpp plugins/action/wait_action.cpp src/register.cpp)
set_target_properties(go2w_foxy_bt_actions PROPERTIES UNITY_BUILD ON)
target_include_directories(go2w_foxy_bt_actions PUBLIC include)
target_compile_definitions(go2w_foxy_bt_actions PRIVATE BT_PLUGIN_EXPORT)
set(dependencies rclcpp rclcpp_action nav2_behavior_tree nav2_msgs nav_msgs geometry_msgs behaviortree_cpp_v3)
ament_target_dependencies(go2w_foxy_bt_actions ${dependencies})
install(TARGETS go2w_foxy_bt_actions LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_libraries(go2w_foxy_bt_actions)
ament_export_dependencies(${dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_bt_action_lifecycle test/test_action_lifecycle.cpp)
  target_include_directories(test_bt_action_lifecycle PRIVATE include)
  ament_target_dependencies(test_bt_action_lifecycle ${dependencies})
endif()
ament_package()
