cmake_minimum_required(VERSION 3.16)
project(go2w_nav2_controller_compat)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(dependencies rclcpp rclcpp_action rcl_interfaces lifecycle_msgs geometry_msgs nav_msgs
  nav2_util nav2_core nav2_msgs nav2_costmap_2d nav_2d_utils nav_2d_msgs pluginlib tf2 tf2_geometry_msgs)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(go2w_foxy_controller_core SHARED src/controller_server.cpp)
target_include_directories(go2w_foxy_controller_core PUBLIC include)
target_compile_definitions(go2w_foxy_controller_core PUBLIC PLUGINLIB__DISABLE_BOOST_FUNCTIONS)
ament_target_dependencies(go2w_foxy_controller_core ${dependencies})
add_executable(controller_server src/main.cpp)
target_include_directories(controller_server PRIVATE include)
ament_target_dependencies(controller_server ${dependencies})
target_link_libraries(controller_server go2w_foxy_controller_core)
install(TARGETS go2w_foxy_controller_core LIBRARY DESTINATION lib)
install(TARGETS controller_server RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_controller_patience test/test_controller_patience.cpp)
  target_include_directories(test_controller_patience PRIVATE include)
  add_test(NAME test_controller_patience COMMAND test_controller_patience)
  add_executable(test_controller_semantics test/test_controller_semantics.cpp)
  target_include_directories(test_controller_semantics PRIVATE include)
  ament_target_dependencies(test_controller_semantics rclcpp rcl_interfaces geometry_msgs nav_msgs tf2 tf2_geometry_msgs)
  add_test(NAME test_controller_semantics COMMAND test_controller_semantics)
endif()
ament_export_include_directories(include)
ament_export_libraries(go2w_foxy_controller_core)
ament_export_dependencies(${dependencies})
ament_package()
