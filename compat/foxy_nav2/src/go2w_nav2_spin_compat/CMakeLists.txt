cmake_minimum_required(VERSION 3.5)
project(go2w_nav2_spin_compat)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_recoveries REQUIRED)
find_package(go2w_nav2_compat REQUIRED)
find_package(nav2_core REQUIRED)
find_package(go2w_nav2_compat_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(go2w_spin_compat SHARED src/spin.cpp src/collision_checker.cpp)
target_include_directories(go2w_spin_compat PUBLIC include)
set(dependencies go2w_nav2_compat rclcpp nav2_recoveries nav2_core go2w_nav2_compat_msgs pluginlib tf2_geometry_msgs)
ament_target_dependencies(go2w_spin_compat ${dependencies})
pluginlib_export_plugin_description_file(nav2_core go2w_nav2_spin_compat.xml)
install(TARGETS go2w_spin_compat LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_libraries(go2w_spin_compat)
ament_export_dependencies(${dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_spin_timing test/test_spin_timing.cpp)
  target_link_libraries(test_spin_timing go2w_spin_compat)
  ament_target_dependencies(test_spin_timing ${dependencies})
endif()
ament_package()
